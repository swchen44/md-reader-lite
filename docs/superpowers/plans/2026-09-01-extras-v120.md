export const CUSTOM_WIDTH_PERCENT_MIN = 20
export const CUSTOM_WIDTH_PERCENT_MAX = 100

export function clampCustomWidthForUnit(
  v: unknown,
  unit: unknown,
): number | null {
  if (v === null || v === undefined || v === '') return null
  const n = typeof v === 'number' ? v : parseFloat(String(v))
  if (!isFinite(n)) return null
  const percent = unit === 'percent'
  const min = percent ? CUSTOM_WIDTH_PERCENT_MIN : CUSTOM_WIDTH_MIN
  const max = percent ? CUSTOM_WIDTH_PERCENT_MAX : CUSTOM_WIDTH_MAX
  return Math.round(Math.min(max, Math.max(min, n)))
}
```

export const PLUGIN_OPTION_DEFAULTS: Record<
  string,
  Record<string, boolean | number | string>
> = {
  TaskLists: { enabled: false, label: false },
  TOC: { maxLevel: 2, listType: 'ul' },
  MultimdTable: {
    multiline: false,
    rowspan: false,
    headerless: false,
    multibody: false,
    autolabel: false,
  },
  Linkify: { fuzzyLink: true, fuzzyIP: false, fuzzyEmail: true },
  Katex: { throwOnError: false, errorColor: '#cc0000' },
  Alert: {
    deep: true,
    infoContainer: true,
    tipContainer: true,
    successContainer: true,
    warningContainer: true,
    dangerContainer: true,
  },
}

export type ResolvedPluginOptions = typeof PLUGIN_OPTION_DEFAULTS

function isRecord(v: unknown): v is Record<string, unknown> {
  return typeof v === 'object' && v !== null && !Array.isArray(v)
}

function sanitize(
  plugin: string,
  key: string,
  value: unknown,
  defaultValue: boolean | number | string,
): boolean | number | string {
  if (typeof defaultValue === 'boolean') {
    return value === true || value === false ? value : defaultValue
  }
  if (plugin === 'TOC' && key === 'maxLevel') {
    const n = Number(value)
    return Number.isInteger(n) && n >= 1 && n <= 6 ? n : defaultValue
  }
  if (plugin === 'TOC' && key === 'listType') {
    return value === 'ul' || value === 'ol' ? value : defaultValue
  }
  if (plugin === 'Katex' && key === 'errorColor') {
    return typeof value === 'string' && /^#[0-9a-f]{6}$/i.test(value)
      ? value
      : defaultValue
  }
  return defaultValue
}

export function resolvePluginOptions(
  userOptions: unknown,
): ResolvedPluginOptions {
  const user = isRecord(userOptions) ? userOptions : {}
  const out: ResolvedPluginOptions = {}
  Object.keys(PLUGIN_OPTION_DEFAULTS).forEach(name => {
    const defaults = PLUGIN_OPTION_DEFAULTS[name]
    const u = isRecord(user[name])
      ? (user[name] as Record<string, unknown>)
      : {}
    const resolved: Record<string, boolean | number | string> = {}
    Object.keys(defaults).forEach(key => {
      resolved[key] = sanitize(name, key, u[key], defaults[key])
    })
    out[name] = resolved
  })
  return out
}

export function tocIncludeLevel(maxLevel: unknown): number[] {
  const n = Number(maxLevel)
  const max = Number.isInteger(n) ? Math.min(6, Math.max(1, n)) : 2
  return Array.from({ length: max }, (_, i) => i + 1)
}

export function diffPluginOptions(
  full: ResolvedPluginOptions,
): Record<string, Record<string, unknown>> {
  const diff: Record<string, Record<string, unknown>> = {}
  Object.keys(PLUGIN_OPTION_DEFAULTS).forEach(name => {
    const defaults = PLUGIN_OPTION_DEFAULTS[name]
    Object.keys(defaults).forEach(key => {
      if (full[name] && full[name][key] !== defaults[key]) {
        ;(diff[name] = diff[name] || {})[key] = full[name][key]
      }
    })
  })
  return diff
}
```
